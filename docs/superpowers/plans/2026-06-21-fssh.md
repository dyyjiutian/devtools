# fssh Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Create `terminal/ssh/fssh` (fzf-driven SSH + tmux picker with terminal cleanup) and `terminal/ssh/setup.sh` (SSH client config optimizer).

**Architecture:** Two independent bash scripts under `terminal/ssh/`. `setup.sh` writes a managed config block to `~/.ssh/config`. `fssh` reads that config, drives two fzf pickers, then SSH-es into the selected tmux session; a `trap` fires `reset_terminal()` on any exit to undo remote tmux's mouse escape sequences.

**Tech Stack:** bash (3.2+), fzf, ssh, tmux (remote), stty/tput for terminal reset.

## Global Constraints

- Bash 3.2+ compatible (macOS ships 3.2) — no `declare -A`, no `mapfile`
- `set -euo pipefail` at top of every script
- Color functions match project pattern: `info()`, `success()`, `warn()`, `error()`
- `--help` and `--dry-run` flags on every script
- Files go under `terminal/ssh/`; `fssh` has no `.sh` extension (it's a binary-like tool)

- [ ] **Step 2: Write `terminal/ssh/setup.sh`**

```bash
cat > terminal/ssh/setup.sh << 'SCRIPT'
#!/usr/bin/env bash
set -euo pipefail

info()    { echo -e "\033[34m[INFO]\033[0m $1"; }
success() { echo -e "\033[32m[OK]\033[0m $1"; }
warn()    { echo -e "\033[33m[WARN]\033[0m $1"; }
error()   { echo -e "\033[31m[ERROR]\033[0m $1"; exit 1; }

SSH_CONFIG="${HOME}/.ssh/config"
SOCKETS_DIR="${HOME}/.ssh/sockets"
MANAGED_START="# === fssh managed: SSH client optimization ==="
MANAGED_END="# === fssh managed end ==="

show_help() {
    cat <<'EOF'
SSH Client Configuration Setup — optimize connections for fssh

Writes the following to ~/.ssh/config (prepended, idempotent):
  ControlMaster auto           — reuse TCP connections to the same host
  ControlPath ~/.ssh/sockets/  — socket directory (created at 700)
  ControlPersist 10m           — keep socket alive 10 min after last session
  ServerAliveInterval 15       — heartbeat every 15 s
  ServerAliveCountMax 3        — disconnect after 3 missed heartbeats (~45 s)
  ConnectTimeout 10            — fail fast on unreachable hosts

Usage:
  ./setup.sh              Apply configuration
  ./setup.sh --dry-run    Preview changes without applying
  ./setup.sh --help       Show this help

To stop a persisted multiplexed connection:
  ssh -O stop <host>
EOF
    exit 0
}

DRY_RUN=false

while [[ $# -gt 0 ]]; do
    case $1 in
        --help|-h) show_help ;;
        --dry-run|-d) DRY_RUN=true; shift ;;
        *) error "Unknown option: $1. Run ./setup.sh --help for usage." ;;
    esac
done

[[ "$DRY_RUN" == "true" ]] && warn "Dry-run mode: no files will be modified"
echo

# ── Managed block content (uses literal ~ so SSH expands it, not shell) ──
MANAGED_BLOCK="${MANAGED_START}
Host *
    # Connection multiplexing: reuse existing handshake for same host
    ControlMaster auto
    ControlPath ~/.ssh/sockets/%r@%h:%p
    ControlPersist 10m

    # Keepalive: heartbeat every 15 s, disconnect after 3 failures (~45 s)
    ServerAliveInterval 15
    ServerAliveCountMax 3

    # Fail fast on unreachable hosts
    ConnectTimeout 10
${MANAGED_END}"

# ── Create sockets directory ──
if [[ ! -d "$SOCKETS_DIR" ]]; then
    info "Creating ${SOCKETS_DIR} (chmod 700)"
    if [[ "$DRY_RUN" == "false" ]]; then
        mkdir -p "$SOCKETS_DIR"
        chmod 700 "$SOCKETS_DIR"
        success "Created ${SOCKETS_DIR}"
    else
        echo "  [dry-run] mkdir -p ${SOCKETS_DIR} && chmod 700 ${SOCKETS_DIR}"
    fi
else
    success "${SOCKETS_DIR} already exists"
fi

# ── Check for existing managed block (idempotent) ──
if [[ -f "$SSH_CONFIG" ]] && grep -qF "$MANAGED_START" "$SSH_CONFIG" 2>/dev/null; then
    success "~/.ssh/config already contains fssh managed block — skipping"
    exit 0
fi

# ── Dry-run: show what would be written ──
if [[ "$DRY_RUN" == "true" ]]; then
    info "Would prepend to ${SSH_CONFIG}:"
    echo ""
    echo "$MANAGED_BLOCK"
    echo ""
    exit 0
fi

# ── Backup existing config ──
if [[ -f "$SSH_CONFIG" ]]; then
    BACKUP="${SSH_CONFIG}.bak.$(date +%Y%m%d%H%M%S)"
    info "Backing up ${SSH_CONFIG} → ${BACKUP}"
    cp "$SSH_CONFIG" "$BACKUP"
fi

# ── Prepend managed block ──
info "Writing SSH client config to ${SSH_CONFIG}"
TMP=$(mktemp)
printf '%s\n\n' "$MANAGED_BLOCK" > "$TMP"
if [[ -f "$SSH_CONFIG" ]]; then
    cat "$SSH_CONFIG" >> "$TMP"
fi
mv "$TMP" "$SSH_CONFIG"
chmod 600 "$SSH_CONFIG"

echo
success "SSH client configuration applied"
info "ControlMaster sockets → ${SOCKETS_DIR}"
info "Stop a persisted connection: ssh -O stop <host>"
SCRIPT
chmod +x terminal/ssh/setup.sh
```

### Task 2: `terminal/ssh/fssh` — fzf SSH + tmux session picker

**Files:**
- Create: `terminal/ssh/fssh`

**Interfaces:**
- Consumes: `~/.ssh/config` Host entries (from existing SSH config)
- Produces: an interactive `ssh -t <host> "tmux new-session -As <session>"` call; `reset_terminal()` called on any exit

- [ ] **Step 1: Write `terminal/ssh/fssh`**

```bash
cat > terminal/ssh/fssh << 'SCRIPT'
#!/usr/bin/env bash
set -euo pipefail

info()    { echo -e "\033[34m[INFO]\033[0m $1" >&2; }
success() { echo -e "\033[32m[OK]\033[0m $1" >&2; }
warn()    { echo -e "\033[33m[WARN]\033[0m $1" >&2; }
error()   { echo -e "\033[31m[ERROR]\033[0m $1" >&2; exit 1; }

# ── Terminal cleanup ──────────────────────────────────────────────────────────
# When remote tmux has `set -g mouse on`, it sends \033[?1000h (and related)
# to enable mouse reporting in the local terminal. On abnormal SSH disconnect
# these sequences are never cancelled, causing Ghostty to misinterpret mouse
# clicks as escape sequences. This trap fires on any exit (normal or signal).
reset_terminal() {
    printf '\033[?1000l'   # disable X10 mouse click reporting
    printf '\033[?1002l'   # disable mouse drag (button-event) reporting
    printf '\033[?1006l'   # disable SGR extended mouse mode
    stty sane 2>/dev/null || true
    tput rmcup 2>/dev/null || true
}
trap reset_terminal EXIT INT TERM

# ── Help ─────────────────────────────────────────────────────────────────────
show_help() {
    cat <<'EOF'
fssh — fuzzy SSH + tmux session picker for Ghostty

Usage:
  fssh              Interactive: pick SSH host, then tmux session
  fssh --dry-run    Show the ssh command that would run, without connecting
  fssh --help       Show this help

Flow:
  1. Reads ~/.ssh/config for Host entries (wildcards excluded)
  2. fzf picker → select host
  3. SSH to host, list tmux sessions
  4. fzf picker → select session, or "[新建会话]" to create a new one
  5. ssh -t <host> "tmux new-session -As <session>"

On any exit (normal or network disconnect), local terminal mouse state is
reset automatically so Ghostty mouse events work correctly again.

Requirements: fzf, ssh
EOF
    exit 0
}

# ── Flags ─────────────────────────────────────────────────────────────────────
DRY_RUN=false

while [[ $# -gt 0 ]]; do
    case $1 in
        --help|-h)    show_help ;;
        --dry-run|-d) DRY_RUN=true; shift ;;
        *) error "Unknown option: $1. Run fssh --help for usage." ;;
    esac
done

# ── Dependency check ──────────────────────────────────────────────────────────
check_deps() {
    local missing=()
    command -v fzf &>/dev/null || missing+=("fzf")
    command -v ssh &>/dev/null || missing+=("ssh")
    if [[ ${#missing[@]} -gt 0 ]]; then
        error "Missing required tools: ${missing[*]}\nInstall: brew install ${missing[*]}"
    fi
}

# ── SSH config parser ─────────────────────────────────────────────────────────
# Extracts Host entries, skipping wildcards (* and ?)
parse_ssh_hosts() {
    local config="${HOME}/.ssh/config"
    [[ -f "$config" ]] || error "~/.ssh/config not found. Add Host entries to use fssh."
    grep -E '^[[:space:]]*Host[[:space:]]' "$config" \
        | awk '{for(i=2;i<=NF;i++) print $i}' \
        | grep -v '[*?]' \
        | sort -u
}

# ── fzf host picker ───────────────────────────────────────────────────────────
pick_host() {
    local hosts
    hosts=$(parse_ssh_hosts)
    [[ -n "$hosts" ]] || error "No hosts found in ~/.ssh/config (only wildcard entries exist)."
    printf '%s\n' "$hosts" \
        | fzf --prompt="SSH Host ❯ " --height=40% --reverse --border --no-multi
}

# ── Remote tmux session lister ────────────────────────────────────────────────
list_remote_sessions() {
    local host="$1"
    # Single SSH hop; suppresses "no server running" errors
    ssh "$host" "tmux list-sessions -F '#{session_name}' 2>/dev/null" 2>/dev/null \
        | grep -v '^$' || true
}

# ── fzf session picker ────────────────────────────────────────────────────────
NEW_SESSION_LABEL="[新建会话]"

pick_session() {
    local host="$1"
    local sessions
    sessions=$(list_remote_sessions "$host")

    local entries
    if [[ -n "$sessions" ]]; then
        entries=$(printf '%s\n%s\n' "$sessions" "$NEW_SESSION_LABEL")
    else
        entries="$NEW_SESSION_LABEL"
    fi

    local choice
    choice=$(printf '%s\n' "$entries" \
        | fzf --prompt="Tmux Session ❯ " --height=40% --reverse --border --no-multi)

    if [[ "$choice" == "$NEW_SESSION_LABEL" ]]; then
        local name
        printf 'Session name [main]: ' >&2
        read -r name
        echo "${name:-main}"
    else
        echo "$choice"
    fi
}

# ── Main ──────────────────────────────────────────────────────────────────────
check_deps

HOST=$(pick_host)
if [[ -z "$HOST" ]]; then
    warn "No host selected."
    exit 0
fi

SESSION=$(pick_session "$HOST")
if [[ -z "$SESSION" ]]; then
    warn "No session selected."
    exit 0
fi

CMD="ssh -t ${HOST} tmux new-session -As ${SESSION}"

if [[ "$DRY_RUN" == "true" ]]; then
    info "Would run: ${CMD}"
    exit 0
fi

success "→ ${HOST}  session: ${SESSION}"
# Do not use exec — trap must fire on exit to clean up terminal state
ssh -t "$HOST" "tmux new-session -As $(printf '%s' "$SESSION" | sed "s/'/'\\\\''/g; s/^/'/; s/$/'/")"
SCRIPT
chmod +x terminal/ssh/fssh
```
